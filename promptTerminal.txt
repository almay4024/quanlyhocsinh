# 1. Tự động kiểm tra & làm sạch .git nếu bị lỗi HEAD / bad object trong OneDrive
git log -1 2>$null | Out-Null
if ($LASTEXITCODE -ne 0) {
    cmd /c "if exist .git (attrib -r -h -s .git & attrib -r -h -s .git\* /s /d & rmdir /s /q .git)"
    git init
    git branch -M main
}
if (-not (git remote | Select-String "origin")) {
    git remote add origin https://github.com/almay4024/quanlyhocsinh.git
} else {
    git remote set-url origin https://github.com/almay4024/quanlyhocsinh.git
}

# 2. Lưu lên GitHub & cập nhật lên web Firebase
git add .
git commit -m "chore: cap nhat phien ban moi ($(Get-Date -Format 'dd/MM/yyyy HH:mm'))"
git push -u origin main --force
firebase deploy --only hosting
