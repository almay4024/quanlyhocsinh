Ảnh tải lên được đặt tên tự động theo quy tắc {Mã_HS}_{Họ_Tên}_{Lớp}_{ID4}.jpg (hoặc CHUAMA_... nếu chưa có mã HS). Khi đổi tên hoặc cập nhật mã HS trên web, file trên Drive tự động đổi tên theo mà vẫn giữ nguyên link cũ.

📋 Xem mã Google Apps Script mẫu (Copy dán vào script.google.com) 
function doPost(e) {
  var data = JSON.parse(e.postData.contents || '{}');
  if (data.action === 'uploadStudentPhoto') {
    var folder = data.folderId ? DriveApp.getFolderById(data.folderId) : getOrCreateFolder_('LopHocVui_AnhHocSinh');
    if (data.oldFileId) { try { DriveApp.getFileById(data.oldFileId).setTrashed(true); } catch (err) {} }
    var parts = String(data.dataUrl || '').split(',');
    var mime = (parts[0].match(/:(.*?);/) || [,'image/jpeg'])[1];
    var bytes = Utilities.base64Decode(parts[1] || '');
    var file = folder.createFile(Utilities.newBlob(bytes, mime, data.fileName || 'student.jpg'));
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return jsonOut_({ ok: true, fileId: file.getId(), fileName: file.getName() });
  }
  if (data.action === 'renameStudentPhoto' && data.fileId && data.newFileName) {
    var f = DriveApp.getFileById(data.fileId);
    f.setName(data.newFileName);
    return jsonOut_({ ok: true, fileId: f.getId(), fileName: f.getName() });
  }
  return jsonOut_({ ok: false });
}
function getOrCreateFolder_(name) {
  var it = DriveApp.getFoldersByName(name);
  return it.hasNext() ? it.next() : DriveApp.createFolder(name);
}
function jsonOut_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
